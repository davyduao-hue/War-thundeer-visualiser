cmake_minimum_required(VERSION 3.20)
project(WTArmourVision VERSION 0.4.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(core INTERFACE)
target_include_directories(core INTERFACE src third_party)
enable_testing()
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE core)
add_test(NAME core_tests COMMAND core_tests ${CMAKE_CURRENT_SOURCE_DIR}/data/database.json)
add_executable(telemetry_tests tests/telemetry_tests.cpp)
target_link_libraries(telemetry_tests PRIVATE core)
add_test(NAME telemetry_tests COMMAND telemetry_tests)
add_executable(regression_tests tests/regression_tests.cpp)
target_link_libraries(regression_tests PRIVATE core)
add_test(NAME regression_tests COMMAND regression_tests)
if(WIN32)
 file(READ "${CMAKE_CURRENT_SOURCE_DIR}/data/database.json" WT_DATABASE_JSON)
 configure_file(src/embedded.hpp.in "${CMAKE_CURRENT_BINARY_DIR}/embedded.hpp" @ONLY)
 set(IMGUI third_party/imgui)
 add_executable(WTArmourVision WIN32 src/windows.cpp ${IMGUI}/imgui.cpp ${IMGUI}/imgui_draw.cpp ${IMGUI}/imgui_tables.cpp ${IMGUI}/imgui_widgets.cpp ${IMGUI}/backends/imgui_impl_win32.cpp ${IMGUI}/backends/imgui_impl_dx11.cpp)
 target_include_directories(WTArmourVision PRIVATE ${IMGUI} ${IMGUI}/backends ${CMAKE_CURRENT_BINARY_DIR})
 target_compile_definitions(WTArmourVision PRIVATE UNICODE _UNICODE NOMINMAX)
 target_link_libraries(WTArmourVision PRIVATE core d3d11 dxgi d3dcompiler dwmapi winhttp psapi shell32)
 if(MINGW)
  target_link_options(WTArmourVision PRIVATE -municode -static -static-libgcc -static-libstdc++)
 endif()
 if(MSVC)
  set_property(TARGET WTArmourVision core_tests telemetry_tests regression_tests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
 endif()
 add_test(NAME windows_ui_smoke COMMAND WTArmourVision --smoke-test)
 add_test(NAME windows_recovery_smoke COMMAND WTArmourVision --smoke-test-recovery)
 set_tests_properties(windows_ui_smoke windows_recovery_smoke PROPERTIES TIMEOUT 30 RUN_SERIAL TRUE)
 install(TARGETS WTArmourVision RUNTIME DESTINATION .)
 install(DIRECTORY data DESTINATION .)
 install(FILES README.md LICENSE ATTRIBUTION.md DESTINATION .)
 install(FILES third_party/json.LICENSE third_party/imgui/LICENSE.txt DESTINATION licences)
endif()
